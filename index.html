const SPREADSHEET_ID = "1atBf7rc9E_NqLBz7zOpkVQbmIsVXMrZU"; const SHEET_NAME = "Sheet1"; // COLUMNS const COL_NAME = 1; // A const COL_ITEM = 9; // I const COL_PRICE = 10; // J const COL_LOCATION = 11; // K const COL_QUANTITY = 15; // O // Hidden technical sheet used to make every scan exactly-once. const TX_SHEET_NAME = "_InventoryTransactions"; // ========================================== // GET SHEET // ========================================== function getSheet() { const ss = SpreadsheetApp.openById( SPREADSHEET_ID ); const sheet = ss.getSheetByName( SHEET_NAME ); if (!sheet) { throw new Error( "Sheet not found" ); } return sheet; } // ========================================== // NORMALIZE ITEM ID // ========================================== function normalizeItem(value) { return String(value || "") .replace("#", "") .trim(); } // ========================================== // FIND ITEM // ========================================== function findItem(itemID) { const sheet = getSheet(); const data = sheet.getDataRange() .getValues(); const item = normalizeItem(itemID); for ( let i = 0; i < data.length; i++ ) { const sheetItem = normalizeItem( data[i][COL_ITEM - 1] ); if ( sheetItem === item && item !== "" ) { return { found: true, sheet: sheet, row: i + 1, data: data[i] }; } } return { found: false }; } // ========================================== // READ QUANTITY // // IMPORTANT: // If item already has a Location and // O is blank, actual Quantity = 1. // // If Location is blank, item has not // yet been inventoried. // ========================================== function readQuantity( location, quantityValue ) { const loc = String(location || "") .trim(); if (!loc) { return 0; } if ( quantityValue === "" || quantityValue === null ) { return 1; } const quantity = Number(quantityValue); if ( isNaN(quantity) || quantity < 1 ) { return 1; } return quantity; } // ========================================== // WRITE QUANTITY // // 0 = blank // 1 = blank // 2+ = actual number // ========================================== function writeQuantity( sheet, row, quantity ) { const cell = sheet.getRange( row, COL_QUANTITY ); if (quantity <= 1) { cell.clearContent(); } else { cell.setValue( quantity ); } } // ========================================== // GET PRODUCT DATA // ========================================== function buildProductData(found) { const row = found.data; const item = normalizeItem( row[COL_ITEM - 1] ); const name = String( row[COL_NAME - 1] || "" ).trim(); const price = row[COL_PRICE - 1]; const location = String( row[COL_LOCATION - 1] || "" ).trim(); const quantity = readQuantity( location, row[COL_QUANTITY - 1] ); return { item: item, name: name, price: price, location: location, quantity: quantity }; } // ========================================== // TRANSACTION / EXACTLY-ONCE HELPERS // ========================================== function getTransactionSheet() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(TX_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(TX_SHEET_NAME); sheet.getRange(1, 1, 1, 6).setValues([[ "Transaction ID", "Item", "Location", "Status", "Result JSON", "Created" ]]); sheet.setFrozenRows(1); sheet.hideSheet(); } return sheet; } function findTransaction(txId) { const id = String(txId || "").trim(); if (!id) return null; const sheet = getTransactionSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return null; const match = sheet.getRange(2, 1, lastRow - 1, 1) .createTextFinder(id) .matchEntireCell(true) .findNext(); if (!match) return null; const row = match.getRow(); const values = sheet.getRange(row, 1, 1, 6).getValues()[0]; return { sheet: sheet, row: row, id: String(values[0] || ""), item: String(values[1] || ""), location: String(values[2] || ""), status: String(values[3] || ""), resultJson: String(values[4] || "") }; } function parseStoredResult(value) { try { return JSON.parse(String(value || "")); } catch (e) { return null; } } function inventoryState(found) { const location = String( found.sheet.getRange(found.row, COL_LOCATION).getValue() || "" ).trim(); const qValue = found.sheet.getRange(found.row, COL_QUANTITY).getValue(); return { location: location, quantity: readQuantity(location, qValue) }; } function sameInventoryState(found, location, quantity) { const state = inventoryState(found); return ( state.location === String(location || "").trim() && state.quantity === Number(quantity) ); } function createPendingTransaction(txId, item, location, result) { const sheet = getTransactionSheet(); sheet.appendRow([ txId, item, location, "PENDING", JSON.stringify(result), new Date() ]); SpreadsheetApp.flush(); return { sheet: sheet, row: sheet.getLastRow() }; } function markTransactionDone(tx, result) { tx.sheet.getRange(tx.row, 4, 1, 2).setValues([[ "DONE", JSON.stringify(result) ]]); SpreadsheetApp.flush(); } // ========================================== // SCAN ITEM // // LOGIC: // // no previous location: // FIRST // Quantity = 1 // // same location: // ADD // Quantity + 1 // // different location: // MOVE // Quantity DOES NOT change // ========================================== function updateLocation(itemID, newLocation, transactionId) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const item = normalizeItem(itemID); const location = String(newLocation || "").trim(); const txId = String(transactionId || "").trim(); if (!item || !location || !txId) { return { success: false, message: "Missing item, location, or transaction ID" }; } const existingTx = findTransaction(txId); if (existingTx) { const stored = parseStoredResult(existingTx.resultJson); if ( existingTx.item !== item || existingTx.location !== location ) { return { success: false, errorType: "TRANSACTION_MISMATCH", message: "Transaction ID mismatch" }; } if (existingTx.status === "DONE" && stored) { stored.replayed = true; stored.verified = true; return stored; } if (existingTx.status === "PENDING" && stored) { const foundPending = findItem(item); if (!foundPending.found) { return { success: false, errorType: "ITEM_NOT_FOUND", message: "ITEM NOT FOUND" }; } // Write already happened; response was lost. if ( sameInventoryState( foundPending, stored.newLocation, stored.quantity ) ) { stored.replayed = true; stored.verified = true; markTransactionDone(existingTx, stored); return stored; } // Transaction was registered but write did not happen. if ( sameInventoryState( foundPending, stored.previousLocation, stored.previousQuantity ) ) { foundPending.sheet .getRange(foundPending.row, COL_LOCATION) .setValue(stored.newLocation); writeQuantity( foundPending.sheet, foundPending.row, stored.quantity ); SpreadsheetApp.flush(); if ( !sameInventoryState( foundPending, stored.newLocation, stored.quantity ) ) { return { success: false, errorType: "VERIFY_FAILED", message: "Inventory write could not be verified", transactionId: txId }; } stored.replayed = true; stored.verified = true; markTransactionDone(existingTx, stored); return stored; } return { success: false, errorType: "TRANSACTION_UNCERTAIN", message: "Transaction state is uncertain. Do not rescan this item.", transactionId: txId }; } } const found = findItem(item); if (!found.found) { return { success: false, errorType: "ITEM_NOT_FOUND", message: "ITEM NOT FOUND" }; } const product = buildProductData(found); const oldLocation = product.location; const oldQuantity = product.quantity; let action = ""; let locationChanged = false; let newQuantity = oldQuantity; if (!oldLocation) { action = "FIRST"; newQuantity = 1; } else if (oldLocation === location) { action = "ADD"; newQuantity = oldQuantity + 1; } else { action = "MOVE"; locationChanged = true; newQuantity = oldQuantity; } const result = { success: true, verified: false, replayed: false, transactionId: txId, action: action, item: product.item, name: product.name, price: product.price, previousLocation: oldLocation, newLocation: location, locationChanged: locationChanged, previousQuantity: oldQuantity, quantity: newQuantity }; // Register the operation BEFORE changing inventory. const pending = createPendingTransaction( txId, item, location, result ); found.sheet .getRange(found.row, COL_LOCATION) .setValue(location); writeQuantity( found.sheet, found.row, newQuantity ); SpreadsheetApp.flush(); // Read K and O back from the sheet. if (!sameInventoryState(found, location, newQuantity)) { return { success: false, errorType: "VERIFY_FAILED", message: "Inventory write could not be verified", transactionId: txId }; } result.verified = true; markTransactionDone( { sheet: pending.sheet, row: pending.row }, result ); return result; } finally { lock.releaseLock(); } } // ========================================== // LOOKUP ITEM // READ ONLY // ========================================== function lookupItem(itemID) { const found = findItem(itemID); if (!found.found) { return { success: false, errorType: "ITEM_NOT_FOUND", message: "ITEM NOT FOUND" }; } const product = buildProductData(found); return { success: true, item: product.item, name: product.name, price: product.price, location: product.location, quantity: product.quantity }; } // ========================================== // SEARCH BY PRODUCT NAME // ========================================== function searchByName(query) { const search = String(query || "") .trim() .toLowerCase(); if (!search) { return { success: false, message: "Enter product name" }; } const sheet = getSheet(); const data = sheet.getDataRange() .getValues(); const results = []; for ( let i = 1; i < data.length; i++ ) { const name = String( data[i][COL_NAME - 1] || "" ).trim(); if (!name) { continue; } if ( name.toLowerCase() .includes(search) ) { const item = normalizeItem( data[i][COL_ITEM - 1] ); if (!item) { continue; } const location = String( data[i][COL_LOCATION - 1] || "" ).trim(); const quantity = readQuantity( location, data[i][COL_QUANTITY - 1] ); results.push({ item: item, name: name, price: data[i][COL_PRICE - 1], location: location, quantity: quantity }); /* Prevent huge search responses. */ if ( results.length >= 20 ) { break; } } } return { success: true, query: query, count: results.length, results: results }; } // ========================================== // UNDO // // Restores EXACT state from before // the last successful scan. // ========================================== function restoreItem( itemID, previousLocation, previousQuantity ) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const found = findItem(itemID); if (!found.found) { return { success: false, errorType: "ITEM_NOT_FOUND", message: "ITEM NOT FOUND" }; } const oldLocation = String( previousLocation || "" ).trim(); let oldQuantity = Number( previousQuantity ); if ( isNaN(oldQuantity) || oldQuantity < 0 ) { oldQuantity = 0; } const locationCell = found.sheet.getRange( found.row, COL_LOCATION ); // Restore Location if (!oldLocation) { locationCell.clearContent(); } else { locationCell.setValue( oldLocation ); } // Restore Quantity writeQuantity( found.sheet, found.row, oldQuantity ); return { success: true, message: "Undo successful", item: normalizeItem( itemID ), restoredLocation: oldLocation, restoredQuantity: oldQuantity }; } finally { lock.releaseLock(); } } // ========================================== // JSON / JSONP RESPONSE // ========================================== function jsonpResponse( result, callback ) { if (callback) { return ContentService .createTextOutput( callback + "(" + JSON.stringify(result) + ");" ) .setMimeType( ContentService .MimeType .JAVASCRIPT ); } return ContentService .createTextOutput( JSON.stringify(result) ) .setMimeType( ContentService .MimeType .JSON ); } // ========================================== // WEB APP // ========================================== function doGet(e) { try { const action = String( e.parameter.action || "" ).trim(); const callback = String( e.parameter.callback || "" ).trim(); // ====================================== // UPDATE / SCAN // ====================================== if (action === "update") { const item = normalizeItem( e.parameter.item ); const location = String( e.parameter.location || "" ).trim(); let result; if ( !item || !location ) { result = { success: false, message: "Missing item or location" }; } else { result = updateLocation( item, location, String(e.parameter.transactionId || "").trim() ); } return jsonpResponse( result, callback ); } // ====================================== // LOOKUP // ====================================== if (action === "lookup") { const item = normalizeItem( e.parameter.item ); let result; if (!item) { result = { success: false, message: "Missing item" }; } else { result = lookupItem(item); } return jsonpResponse( result, callback ); } // ====================================== // SEARCH BY NAME // ====================================== if (action === "search") { const query = String( e.parameter.query || "" ).trim(); const result = searchByName( query ); return jsonpResponse( result, callback ); } // ====================================== // UNDO // ====================================== if (action === "undo") { const item = normalizeItem( e.parameter.item ); const previousLocation = String( e.parameter.previousLocation || "" ).trim(); const previousQuantity = String( e.parameter.previousQuantity || "0" ).trim(); let result; if (!item) { result = { success: false, message: "Missing item" }; } else { result = restoreItem( item, previousLocation, previousQuantity ); } return jsonpResponse( result, callback ); } // ====================================== // NORMAL PAGE // ====================================== const template = HtmlService .createTemplateFromFile( "index" ); template.location = e.parameter.location || ""; template.item = e.parameter.item || ""; return template .evaluate() .setTitle( "Inventory Scanner" ); } catch(error) { return jsonpResponse( { success: false, message: String( error.message || error ) }, String( e.parameter.callback || "" ) ); } }
